// Temporary coverage-review probe (was ironhorse-262/tests/zz_audit_cov_1.rs; deleted after the run):
// runs each assembled async test262 file in ZZ_DIR on a plain IronHorse realm (no oracle),
// drains jobs, and prints the `__sig` completion latch the assembled prelude records.
#[test]
fn zz_async_plain_host() {
    let dir = std::env::var("ZZ_DIR").expect("ZZ_DIR");
    let mut entries: Vec<_> = std::fs::read_dir(&dir).unwrap().map(|e| e.unwrap().path()).collect();
    entries.sort();
    let out = std::env::var("ZZ_OUT").expect("ZZ_OUT");
    let mut lines = String::new();
    for p in entries {
        let src = std::fs::read_to_string(&p).unwrap();
        let name = p.file_name().unwrap().to_string_lossy().to_string();
        let res = std::thread::Builder::new()
            .stack_size(ironhorse_vm::NATIVE_STACK_BYTES)
            .spawn(move || {
                match ironhorse_compile::compile_atoms_goal(&src, ironhorse_compile::Goal::Script, false) {
                    Err(e) => format!("COMPILE\t{:?}", e.kind),
                    Ok((bc, sy)) => {
                        let names = ironhorse_vm::parse_symbols(&sy);
                        let mut interp = ironhorse_vm::Interp::new();
                        interp.install_test262_host();
                        interp.link_intrinsics(&names);
                        interp.set_source_compiler(std::rc::Rc::new(ironhorse_262::IronhorseSourceCompiler));
                        let o = interp.run(&bc).host_coerced();
                        let sig = interp.global_string("__sig");
                        format!("{}\t{:?}\t{}", if o.completed { "RAN" } else { "HALT" },
                            sig, if o.completed { String::new() } else { format!("{:?}", o.halt).chars().take(300).collect() })
                    }
                }
            })
            .unwrap()
            .join()
            .unwrap_or_else(|_| "PANIC".to_string());
        lines.push_str(&format!("{name}\t{res}\n"));
    }
    std::fs::write(out, lines).unwrap();
}
