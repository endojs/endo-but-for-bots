let err;
try {
  typeof dflt;
} catch(e) { err = e; } finally {
  if (!(err instanceof ReferenceError)) {
    throw `binding is created but not initialized`;
  }
}

import dflt from './index.js';

### @michaelfig (MEMBER) — 2019-11-19T20:18:10Z

In our emulation, there is no temporal dead zone between when a binding is imported and when it is initialized.

This is a [known divergence](https://github.com/Agoric/make-importer/blob/master/doc/divergence.md#imported-variable-have-no-temporal-dead-zone).  

I've downgraded this to `wontfix`.
