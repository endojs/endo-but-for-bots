```js
const canBeMethodName = key => {
  if (typeof key === 'string') {
    return key !== '@@asyncIterator' &&
      key !== 'constructor' &&
      key !== 'then';
  } else if (typeof key === 'symbol') {
    return key === Symbol.asyncIterator ||
      key === Symbol.for('constructor') ||
      key === Symbol.for('then');
  } else {
    return false;
  }
}
```
