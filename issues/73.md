## Description

This seems to be caused by an issue with the evaluator, however we need tests to ensure that given a correct evaluator, isolation is maintained between importers.

**Steps to reproduce**
```js
// test/test-moddir-2.js
/* eslint-disable import/no-extraneous-dependencies */
import { test } from 'tape-promise/tape';
import { makeEvaluators } from '@agoric/evaluate';
import makeModuleTransformer from '@agoric/transform-module';
import * as babelCore from '@babel/core';
import fs from 'fs';
import path from 'path';

import makeImporter, * as mi from '../src';

const readFile = ({ pathname }) => fs.promises.readFile(pathname, 'utf-8');

const protoHandlers = new Map([['file', readFile]]);

const rootUrl = `file://${path.join(__dirname, 'moddir')}`;

test('moddir mod1.js', async t => {
  const transforms = [];
  const { evaluateProgram } = makeEvaluators();
  const importer = makeImporter({
    resolve: mi.makeRootedResolver(rootUrl),
    locate: mi.makeSuffixLocator('.js'),
    retrieve: mi.makeProtocolRetriever(protoHandlers),
    rewrite: mi.makeTransformRewriter(transforms),
    rootLinker: mi.makeEvaluateLinker(evaluateProgram),
  });
  transforms[0] = makeModuleTransformer(babelCore, importer);
  try {
    await importer({ spec: './mod1', url: `${rootUrl}/` }, {});
  } catch (e) {
    t.isNot(e, e, 'unexpected exception');
  } finally {
    t.end();
  }
});

test('moddir mod2.js', async t => {
  const transforms = [];
  const { evaluateProgram } = makeEvaluators();
  const importer = makeImporter({
    resolve: mi.makeRootedResolver(rootUrl),
    locate: mi.makeSuffixLocator('.js'),
    retrieve: mi.makeProtocolRetriever(protoHandlers),
    rewrite: mi.makeTransformRewriter(transforms),
    rootLinker: mi.makeEvaluateLinker(evaluateProgram),
  });
  transforms[0] = makeModuleTransformer(babelCore, importer);
  try {
    await importer({ spec: './mod2', url: `${rootUrl}/` }, {});
  } catch (e) {
    t.isNot(e, e, 'unexpected exception');
  } finally {
    t.end();
  }
});
```
```js
// test/moddir/mod1.js
const global = new Function('return this')();
global.test262 = 262;
```
```js
// test/moddir/mod2.js
const global = new Function('return this')();
if (global.test262) {
  throw 'global leaking between tests';
}
```
Current results:
```
$ node  -r esm test/test-moddir-2.js 
TAP version 13
# moddir mod1.js
# moddir mod2.js
not ok 1 unexpected exception
  ---
    operator: notEqual
    expected: 'global between through tests'
    actual:   'global between through tests'
```
Expected results: no error
```
$ node -r esm test/moddir/mod1.js
$ node -r esm test/moddir/mod2.js
// no error
```
